cmake_minimum_required(VERSION 3.12)
project(mGBAWeb C CXX)
if(NOT EMSCRIPTEN)
  message(FATAL_ERROR "Configure with emcmake cmake -S web -B build-web")
endif()
set(LIBMGBA_ONLY ON)
set(BUILD_GL OFF)
set(BUILD_GLES2 OFF)
set(BUILD_GLES3 OFF)
set(ENABLE_DEBUGGERS OFF)
set(ENABLE_SCRIPTING OFF)
add_compile_definitions(DISABLE_THREADING DISABLE_ANON_MMAP)
add_subdirectory(.. core)
add_executable(mgba-web bridge.c celio-device.cpp)
get_target_property(CORE_DEFINITIONS mgba COMPILE_DEFINITIONS)
target_compile_definitions(mgba-web PRIVATE ${CORE_DEFINITIONS})
target_link_libraries(mgba-web mgba)
set_target_properties(mgba-web PROPERTIES OUTPUT_NAME mgba SUFFIX ".js" LINK_FLAGS "--no-entry -sMODULARIZE=1 -sEXPORT_NAME=createMGBA -sALLOW_MEMORY_GROWTH=1 -sSTACK_SIZE=1048576 -sENVIRONMENT=web,node -sEXPORTED_FUNCTIONS=['_malloc','_free'] -sEXPORTED_RUNTIME_METHODS=['HEAPU8','HEAP16']")
configure_file(index.html index.html COPYONLY)
configure_file(app.js app.js COPYONLY)
foreach(asset rom-library.js skins.js vendor/fflate.min.js vendor/FFLATE-LICENSE.txt vendor/pdf.min.js vendor/pdf.worker.min.js vendor/PDFJS-LICENSE.txt memory-viewer.js celio-serial.js ui.js link-session.js manifest.webmanifest sw.js offline.js screen-effects.js shaders/xbrz-pass0.glsl shaders/xbrz-pass1.glsl shaders/COPYING.txt shaders/README.md styles.css controller.js storage.js help.html)
  configure_file(${asset} ${asset} COPYONLY)
endforeach()
configure_file(../res/mgba-256.png icon-256.png COPYONLY)
configure_file(../res/mgba-512.png icon-512.png COPYONLY)
